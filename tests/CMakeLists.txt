# ----------------------------------------------------------------------------
# tests/
#
# Each test file becomes a single executable. gtest_discover_tests then
# registers every TEST() within it with ctest, so individual cases are
# selectable via -R and reported separately.
# ----------------------------------------------------------------------------

include(GoogleTest)

set(MQCES_UNIT_TESTS
    unit/test_spatial_rank.cpp
    unit/test_spatial_rank_gemm.cpp
    unit/test_spatial_rank_approx.cpp
    unit/test_inverse_rank.cpp
    unit/test_inverse_vz.cpp
    unit/test_inverse_vz_aa.cpp
    unit/test_score.cpp
    unit/test_uncertainty.cpp
    unit/test_parallel_shim.cpp
    unit/test_classify_classic.cpp
    unit/test_classify_v2.cpp
    unit/test_classify_v3.cpp
    unit/test_classify_v4.cpp
    unit/test_nota.cpp
)

foreach(_src IN LISTS MQCES_UNIT_TESTS)
    get_filename_component(_name "${_src}" NAME_WE)
    add_executable(${_name} ${_src})
    target_link_libraries(${_name}
        PRIVATE
            mqces::mqces
            mqces::warnings
            GTest::gtest_main
    )
    gtest_discover_tests(${_name}
        DISCOVERY_TIMEOUT 30
        PROPERTIES LABELS unit
    )
endforeach()

# ---- Integration tests ----------------------------------------------------
# Fixture data is split across two big translation units generated by
# tests/data/generate_fixtures.py. They're compiled into an OBJECT library
# so each integration-test binary can pull them in once.
add_library(mqces_test_fixtures OBJECT
    data/fixtures_data.cpp
    data/expected_scores.cpp
)
target_include_directories(mqces_test_fixtures PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/data")

set(MQCES_INTEGRATION_TESTS
    integration/test_burnup_pipeline.cpp
    integration/test_paper_table1.cpp
    integration/test_sampling_score_drift.cpp
    integration/test_solver_score_parity.cpp
)

foreach(_src IN LISTS MQCES_INTEGRATION_TESTS)
    get_filename_component(_name "${_src}" NAME_WE)
    add_executable(${_name} ${_src})
    target_link_libraries(${_name}
        PRIVATE
            mqces::mqces
            mqces::warnings
            mqces_test_fixtures
            GTest::gtest_main
    )
    gtest_discover_tests(${_name}
        DISCOVERY_TIMEOUT 60
        PROPERTIES LABELS integration
    )
endforeach()
